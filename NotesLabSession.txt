@Controller
public class EmployeeController
{

	@Autowired
	EmployeeService empSvc;
	@QueryMapping
	List <Employee> getEmployees()
	{
		return empSvc.getAll();

	}

	@QueryMapping
	Employee getEmployeeById(@Argument int id)
	{
		return empSvc.getById(id)
	}

	@MutationMapping
	public Employee saveEmployee(@Argument int id,@Argument String empName,@Argument String empAddrs)
	{
		Employee e = new Employee(id,empName,empAddrs)
		return empSvc.saveEmp(e);
	}

	@MutationMapping
	public Employee updateEmployee(@Argument int id,@Argument int salary)
	{

		return empSvc.updateEmployee(id,salary)
	}


}
